\title{\textbf{Ballot words and the Morin Thom polynomials}}
\author{Chern++}
\date{September 2026}

\begin{document}

\begin{abstract}
The Chern coefficients $C(M)$ of the Morin Thom polynomials $\Tp_{A_d}$ are indexed by zero-sum
multisets $M$ of $d$ integers. Let $b(M)$ be the number of orderings of $M$ whose proper partial
sums are nonnegative. We prove, given two geometric inputs that we state precisely, that
$C(M) = b(M)$ whenever $\max M \le 1$. On that sector $b(M)$ is a Kreweras number. We conjecture that
$C(M) \ge b(M)$ for every $M$, with equality exactly on that sector. This implies Rim\'anyi's
positivity conjecture. We prove it, with its equality case, for $d \le 4$. The case $d = 4$ rests
on a new explicit positive product formula for the chamber series together with exact integer-set
inclusions. The conjecture also holds on every exact coefficient we can compute: all $d \le 7$ on large boxes, including $\Tp_{A_7}$ to $\ell = 5$ and every packet of
charge $\le 6$ at $d = 7$.
\end{abstract}

\section{Setting}

By Rim\'anyi's $\ell$-free reduction~\cite{Rim01} and the B\'erczi--Szenes residue formula~\cite{BS12},
\[
  \Tp^{\ell}_{A_d} = \sum_{M} C(M) \prod_{a \in M} c_{\ell + 1 + a},
\]
where $M$ ranges over multisets of $d$ integers summing to $0$, $c_0 = 1$, and $c_j = 0$ for $j < 0$. The
coefficient $C(M)$ does not depend on $\ell$. In chamber coordinates,
\[
  C(M) = \sum_{\alpha} A_{\beta(\alpha)} ,
\]
where the sum is over the distinct orderings $\alpha$ of $M$, the prefix sums
$\beta_j = \alpha_1 + \dots + \alpha_j$ ($j < d$) must be nonnegative, and $F_d = \sum_\beta A_\beta
x^\beta$ is the Morin chamber series. Rim\'anyi's conjecture is $C(M) \ge 0$.

\begin{definition}
A \emph{ballot ordering} of $M$ is an ordering whose proper prefix sums are all nonnegative. Let
$b(M)$ be the number of distinct ballot orderings. Equivalently, $b(M)$ is $C(M)$ computed with the
pure chain series $\prod_{j=1}^{d-1}(1 - x_j)^{-1}$, all of whose coefficients are $1$, in place
of $F_d$.
\end{definition}

The decreasing ordering is always ballot, so $b(M) \ge 1$. The \emph{charge} of $M$ is $\max M$. The
\emph{plane sector} is charge $\le 1$; for $\ell = L - 1$ these are the Chern monomials whose largest
index is at most $\ell + 2$.

\section{The plane sector}

\begin{theorem}\label{thm:plane}
Assume the geometric inputs (I1) and (I3) of \S\ref{sec:inputs}. If $\max M \le 1$ then
$C(M) = b(M)$. Write $M = (1^{d-b}, 1-s_1, \dots, 1-s_b)$ with every $s_i \ge 1$, and let $m_i$ be
the number of $s$'s equal to $i$. Then this common value is the number of noncrossing partitions
of $[d]$ with block sizes $s_1, \dots, s_b$:
\[
  C(M) = b(M) = \frac{d!}{(d-b+1)!\ \prod_i m_i!} .
\]
Independently of (I1) and (I3), the identity $C(M) = b(M)$ is verified by exact computation from
the chamber series on every plane-sector packet with $d \le 7$.
\end{theorem}

\subsection{The geometric inputs}\label{sec:inputs}

Fix $n = d + 1$ and $r \ge 2$.
\begin{itemize}[nosep]
\item Let $C_{n,r}$ be the \emph{curvilinear component} of the punctual Hilbert scheme
  $\mathrm{Hilb}^n_0(\mathbb{C}^r)$. This is the closure of the locus of length-$n$ subschemes
  supported at the origin and isomorphic to $\operatorname{Spec} \mathbb{C}[t]/(t^n)$. It is
  irreducible of dimension $(r-1)d$.
\item Let $V$ be the tautological rank-$n$ bundle, whose fibre at $Z$ is $\mathcal{O}_Z$.
\item Let $E = \ker(V \to \mathcal{O})$ be its rank-$d$ augmentation part, with Chern roots
  $\epsilon_1, \dots, \epsilon_d$.
\end{itemize}
Write the Thom polynomial in the Chern classes $c_i = e_i(x_1, \dots, x_q)$ of formal roots
$x_1, \dots, x_q$, where $q = r + \ell$; taking $r$ large enough makes $q$ exceed every index
that occurs.

\begin{itemize}[leftmargin=3.2em]
\item[\textbf{(I1)}] \emph{Incidence formula.}
  $\displaystyle \Tp^\ell_{A_d} = \int_{C_{n,r}} \prod_{a=1}^{q} \prod_{i=1}^{d} (x_a + \epsilon_i)$.

  This is the curvilinear form of the B\'erczi--Szenes theorem~\cite{BS12}. Reason: a $q$-tuple
  of function germs on $\mathbb{C}^r$ defines an $A_d$ singularity exactly when its zero scheme is
  curvilinear of length $n$. The incidence variety of pairs (curvilinear $Z$, $q$-tuple vanishing on
  $Z$) is a vector bundle over $C_{n,r}$, namely the kernel of evaluation onto $E \otimes \mathbb{C}^q$.
  Its projection to the space of $q$-tuples is generically one-to-one onto the $A_d$ locus, which has
  codimension $qd - (r-1)d = d(\ell+1)$. The pushforward is the displayed integral. In exactly this
  form, with this proof, it is stated in~\cite{FS}.

\item[\textbf{(I2)}] \emph{Extraction.} Let $\lambda = (\lambda_1, \dots, \lambda_s)$ have all parts
  $\le q$ and $s \le d$, and let
  $\nu = \mathrm{sort}\,(q^{d-s}, q - \lambda_1, \dots, q - \lambda_s)$. Then the coefficient of the
  Chern monomial $c_\lambda$ in (I1) is $\int_{C_{n,r}} m_\nu(\epsilon)$, where $m_\nu$ is the
  monomial symmetric function.

  \emph{Proof.} $\prod_{i}\prod_{a} (x_a + \epsilon_i) = \prod_i \sum_{j=0}^{q} e_j(x)\,
  \epsilon_i^{q-j}$. The coefficient of $e_{\lambda_1}(x) \cdots e_{\lambda_s}(x)\, e_0(x)^{d-s}$ is the
  sum over distinct assignments of the indices $\lambda_1, \dots, \lambda_s, 0^{d-s}$ to the $d$ roots,
  that is, $m_\nu(\epsilon)$. \qed

\item[\textbf{(I3)}] \emph{Hyperplane lemma.} $c_d(E) \cap [C_{n,r}] = [C_{n,r-1}]$, where
  $C_{n,r-1}$ is embedded as the subschemes lying in a coordinate hyperplane.

  Reason: the last coordinate $u_r$ gives a section of $E$. It vanishes exactly on the subschemes
  contained in $\{u_r = 0\}$. On the standard curvilinear chart $I = (u_2 - p_2(t), \dots,
  u_r - p_r(t), t^n)$ it is the regular $d$-tuple of coefficients of $p_r$, so the zero locus is
  reduced of the expected codimension $d$. This is stated and proved in this form in~\cite{FS}.

\item[\textbf{(I4)}] \emph{The plane} (classical). For $r = 2$, $C_{n,2} = \mathrm{Hilb}^n_0(\mathbb{C}^2)$
  by Brian\c{c}on's theorem~\cite{Bri77}. Its cohomology is identified by Lehn--Sorger~\cite{LS01}
  with the associated graded of the class algebra of $S_n$ (filtered by $n$ minus the number of
  cycles). Under this identification the elementary symmetric functions of the $\epsilon_i$ go to
  those of the Jucys--Murphy elements $J_k = \sum_{a<k}(a\,k)$. The top degree is spanned by the
  class of the $n$-cycles, normalised by $\int c_d(E) = 1$. So for $|\nu| = d$,
  $\int_{C_{n,2}} m_\nu(\epsilon) = [\sigma]\, m_\nu(J_1, \dots, J_n)$ for any $n$-cycle $\sigma$.
\end{itemize}

\paragraph{Reduction to the plane.} Take $\ell = L - 1$ and a plane-sector $M$, so the Chern monomial
is $c_\lambda$ with $\lambda_i = L + \alpha_i \le L + 1$ for the entries $\alpha_i$ of $M$ with
$L + \alpha_i > 0$. With $q = r + L - 1$, every entry of $\nu$ is at least $q - L - 1 = r - 2$. While
$r > 2$ all entries are positive, so $m_\nu(\epsilon) = (\epsilon_1 \cdots \epsilon_d)\,
m_{\nu - 1}(\epsilon) = c_d(E)\, m_{\nu-1}(\epsilon)$. By (I3) this lowers $r$ by one and every entry
of $\nu$ by one. At $r = 2$, $q = L + 1$ and the entries of $\nu$ are $L + 1 - \lambda_i = 1 - \alpha_i$,
together with $q = 1 - (-L)$ for the entries $\alpha = -L$ that give $c_0$. So $\nu$ is the
multiset $1 - M$, and by (I4)
\[
  C(M) = [\sigma]\ m_{1 - M}(J_1, \dots, J_n).
\]
Everything below is new.

\begin{lemma}\label{lem:monotone}
Let $\sigma(x) = x + 1 \pmod n$. A \emph{minimal monotone factorisation} of $\sigma$ is an expression
$\sigma = (a_1\,b_1)(a_2\,b_2)\cdots(a_{n-1}\,b_{n-1})$ with $a_i < b_i$ and $b_1 \le \dots \le
b_{n-1}$. Its exponent vector is $e_k = \#\{i : b_i = k\}$, with $e_1 = 0$. Then for every
$e \in \mathbb{Z}_{\ge 0}^n$ with $\sum e_k = n - 1$:
\[
  \#\{\text{minimal monotone factorisations of } \sigma \text{ with exponent vector } e\}
  = \begin{cases} 1 & \text{if } D(j) := j - 1 - \sum_{k \le j} e_k \ge 0 \text{ for all } j, \\
                  0 & \text{otherwise.} \end{cases}
\]
\end{lemma}

\begin{proof}
\emph{Necessity.} The $n-1$ transpositions of a minimal factorisation of an $n$-cycle are the edges
of a tree on $[n]$. Those with $b_i \le j$ are edges of a forest on $[j]$, so there are at most
$j - 1$ of them.

\emph{Existence and uniqueness,} by induction on $n$. The case $n = 1$ is trivial. Let $m = e_n$
and let $(a_1\,n)\cdots(a_m\,n)$ be the final $m$ factors, with product $C$. The remaining
factors form a minimal monotone factorisation of $F = \sigma C^{-1}$, a permutation of $[n-1]$ with
exactly $m$ cycles. Since $C^{-1} = (n\ a_1\ a_2 \cdots a_m)$ we have $a_1 = \sigma^{-1}(n) = n-1$ and
\[
  F(x) = x + 1 \ (x \notin \{a_i\}), \qquad F(a_i) = a_{i+1} + 1 \ (i < m), \qquad F(a_m) = 1 .
\]
Because $F \cdot C = \sigma$ is one cycle and $C$ is an $(m+1)$-cycle, each cycle of $F$ contains
exactly one $a_i$. Starting at $a_{i+1} + 1$ and stepping by $+1$, the cycle through $a_i$ must meet
$a_i$ before any other $a_j$. So $a_{i+1} < a_i$ with no $a_j$ in between, and $a_m$ is the least.
Hence $a_1 > a_2 > \dots > a_m$, and $F$ is the product of the increasing cycles on the consecutive
intervals
\[
  I_m = [1, a_m],\quad I_{m-1} = [a_m + 1, a_{m-1}],\quad \dots,\quad I_1 = [a_2 + 1, n - 1].
\]
Transpositions on disjoint intervals commute, so monotone factorisations of $F$ are exactly
shuffles, merged by $b$, of monotone factorisations of the interval cycles. Minimality on each
interval says the exponents inside $I_{m-i}$ sum to $|I_{m-i}| - 1$. Since $D$ changes by
$1 - e_k \le 1$ per step, this means $D(a_{m-i}) = i$, and the ballot condition inside the interval
means $D > i$ strictly after $a_{m-i}$, up to the next cut. So $a_{m-i}$ is the last $j$ with
$D(j) = i$. When $e$ is ballot these last visits exist, are increasing in $i$, and end at
$D(n-1) = m - 1$. So the cut points exist and are forced, and the induction applies to each interval
(each is again an increasing cycle, of smaller length).
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:plane}]
Expanding $m_\nu(J_1, \dots, J_n) = \sum_e \prod_k J_k^{e_k}$ over distinct rearrangements $e$ of
$\nu$ (padded by zeros, $e_1 = 0$ since $J_1 = 0$), the coefficient of $\sigma$ counts pairs
($e$, minimal monotone factorisation with exponent vector $e$). By Lemma~\ref{lem:monotone} this is the
number of ballot vectors $e$. Setting $\alpha_k = 1 - e_{k+1}$ ($k = 1, \dots, d$) turns the condition
$D \ge 0$ into $\alpha_1 + \dots + \alpha_j \ge 0$, and $e$ into an ordering of $M$. So $C(M) = b(M)$.
For the product formula, apply the cycle lemma to $-M$ with a final $-1$ appended: $n$ integers
$\ge -1$ summing to $-1$, exactly one cyclic rotation of which has all proper prefix sums nonnegative.
Reversal identifies these rotations with ballot orderings of $M$, so
$b(M) = \frac1n \cdot \frac{n!}{(d-b+1)!\prod_i m_i!}$. This is Kreweras' count~\cite{Kre72}.
\end{proof}

Lemma~\ref{lem:monotone} is checked exhaustively for $n \le 7$; the totals are the Catalan numbers
$1, 2, 5, 14, 42, 132$. The identity $C(M) = b(M)$ is checked on every plane-sector packet for
$d \le 7$ directly from the chamber series. Both checks are in tier~12 of the test suite.

\begin{remark}
On the plane sector the Morin chamber series is therefore indistinguishable, packet by packet,
from the pure chain series $\prod_j (1 - x_j)^{-1}$. For example, at $d = 7$ the packet
$M = (1,1,1,0,-1,-1,-1)$ has $35$ ballot orderings. Eight of them carry nonzero $A_\beta$, namely
$1, 4, -1, -4, 5, 6, 6, 18$, and these sum to $35$.
\end{remark}

\section{The ballot conjecture}

\begin{conjecture}[ballot]\label{conj:ballot}
For every zero-sum multiset $M$ of $d$ integers, $C(M) \ge b(M)$, with equality if and only if
$\max M \le 1$. Equivalently: the series $F_d - \prod_j (1 - x_j)^{-1}$ has nonnegative packet sums,
and they vanish exactly on the plane sector.
\end{conjecture}

\begin{corollary}
Conjecture~\ref{conj:ballot} implies Rim\'anyi's conjecture, and more: every Chern coefficient of
$\Tp_{A_d}$ is at least $1$ whenever the monomial occurs.
\end{corollary}

This explains a feature of the data that had no explanation: across $d = 4, \dots, 7$ the least Chern
coefficient is $1$, never $0$. It is attained exactly at $M = 0$ and $M = (1^{d-1}, 1-d)$, the two
multisets with $b(M) = 1$ on the plane sector.

\paragraph{Evidence.} Table~\ref{tab:ballot} tests the conjecture on every exact table available.
The data come from the exact sweep of~\texttt{chernpp.boxes}, reconstructed by CRT and verified
against a held-back prime. The level box $\min M \ge -L$ contains every packet of $\Tp^{\ell}$ with
$\ell < L$; the charge box contains every packet with $\max M \le p$. There is no violation, no
plane-sector packet with $C \ne b$, and no off-plane packet with $C = b$. Off the plane sector
$C(M)/b(M) \ge 2$ throughout. The minimum is $2$ at $(2,-1,-1)$ and grows slowly with $d$
($9/4$, $12/5$, $79/30$, $58/21$), attained on the family $(2, 1^{d-4}, -1^{d-2})$ and its
neighbours.

\paragraph{What does not work.}
\begin{itemize}[nosep]
\item \emph{Factorial weights.} $C(p, -1^p) = p!$, which suggests weighting each ballot word by
  $\prod_{\alpha_i > 0} \alpha_i!$. That bound fails already at $d = 3$.
\item \emph{Fusion.} Replacing a $2$ by two $1$s and counting plane-sector ballot words of the
  longer multiset gives some families exactly. For example, $C(2, 0^{d-2}, -2) = \binom{d+1}{3} +
  \binom{d}{2}$, where both terms are plane-sector counts. But the residual is negative elsewhere.
\item \emph{Local step weights.} Keep weight $1$ on steps $\le 1$ and give each step of size $2$ a
  weight depending on local data: its height, together with the neighbouring steps, the number of
  remaining steps, or the next return. Fitted exactly on the packets with one $2$, every such model
  that is consistent within a single $d$ is inconsistent across $d$.
\item \emph{Formal Jucys--Murphy extension.} Theorem~\ref{thm:plane} reads $C(M) = \sum_{\alpha}
  [\sigma] \prod_k J_{k+1}^{1-\alpha_k}$ over ballot $\alpha$. At charge $2$ this needs $J_k^{-1}$,
  which exists for $k = 2, 3$ (their eigenvalues, the contents of boxes $2$ and $3$, are nonzero).
  Where it can be evaluated, it returns the wrong values, some of them fractional ($C(2,0,-2) = 7$
  against $23/2$). At $d = 4$ the needed inverses do not exist.
\item \emph{Individual coefficients.} No bound on single $A_\beta$ can work: they are unbounded
  below ($-3622$ at $d = 7$), and the smallest symmetrisation that clears all negatives grows with
  $d$~\cite{Rep}.
\end{itemize}

\section{The conjecture for $d \le 4$}

At fixed $d$ the plane sector is finite: its entries lie in $[1-d, 1]$. So the equality part of
Conjecture~\ref{conj:ballot} at fixed $d$ is a finite check. It needs neither
Theorem~\ref{thm:plane} nor its geometric inputs. The content is the inequality over infinitely
many $M$, together with strictness off the plane sector. We prove both for $d \le 3$, directly from
the chamber series.

In chamber coordinates write $x = x_1$, $y = x_2$. The chamber series is a power series in these
variables, and its exponents $\beta \ge 0$ are exactly the prefix-sum vectors of ballot orderings.
For $d \le 3$ the multidegree numerator is trivial, and the B\'erczi--Szenes formula~\cite{BS12}
gives
\[
  F_2 = \frac{1 - x}{1 - 2x}, \qquad
  F_3 = \frac{(1-x)(1-y)(1-xy)}{(1-2x)(1-2xy)(1-y-xy)} .
\]
These are the stored chamber algebras of this project, checked in tier~15. The factorisation of
the numerator of $F_3$ is a direct expansion.

\begin{theorem}\label{thm:small}
For $d \le 3$ and every zero-sum $M$: $C(M) \ge b(M)$, with equality if and only if
$\max M \le 1$. Moreover $C(M) \ge b(M) + 1$ whenever $\max M \ge 2$.
\end{theorem}

\begin{proof}
\emph{$d = 2$.} $F_2 = 1 + x/(1-2x)$, so $C(k,-k) = 2^{k-1}$ for $k \ge 1$ and $C(0,0) = 1$, while
$b = 1$ throughout.

\emph{$d = 3$.} Each factor of $F_3$ has the form $\frac{1-u}{1-u-v} = 1 + \frac{v}{1-u-v}$ with $u, v$
monomials with nonnegative coefficients:
\[
  F_3 = (1 + U)(1 + V)(1 + W), \quad U = \frac{x}{1-2x},\ V = \frac{xy}{1-2xy},\ W = \frac{xy}{1-y-xy}.
\]
All of $U, V, W$ have nonnegative coefficients, so $F_3 \ge 1 + U + V + W + UV + UW$ coefficientwise.
Compare with
\[
  Z_3 = 1 + \frac{x}{(1-x)(1-y)} + \frac{xy}{1-xy},
\]
whose coefficient at $x^i y^j$ is $1$ at $(0,0)$, $0$ for $i = 0 < j$, and $1 + [i = j]$ for $i \ge 1$.

(i) $V \ge xy/(1-xy)$, since $2^{k-1} \ge 1$.

(ii) $U + W + UW \ge x/((1-x)(1-y))$. Write $[x^iy^j]$ for a coefficient. Then $[x^i]U = 2^{i-1}$
covers $j = 0$. $W = \sum_{m \ge 0} xy\,(y + xy)^m$ has $[x^{1+k}y^{1+m}]W = \binom mk$, which covers
$1 \le i \le j$. And $[x^i y^j]\,UW \ge [x^{i-1}]U \cdot [x y^{j}]W = 2^{i-2}$ covers $1 \le j < i$.

So $F_3 \ge Z_3$ coefficientwise. Next, the packet sums of $Z_3$ are $b(M)$. A ballot ordering $\alpha$
of $M = (a,b,c)$ has $\beta = (\alpha_1, \alpha_1 + \alpha_2)$.
- It lands on the zero set $\{i = 0 < j\}$ exactly when $\alpha = (0, k, -k)$, which requires
  $M = (k, 0, -k)$.
- It lands on the diagonal $i = j \ge 1$ exactly when $\alpha = (k, 0, -k)$, with the same $M$.

Both occur exactly for $M = (k, 0, -k)$, $k \ge 1$, and then they cancel. So the $Z_3$-packet sum is
the number of ballot orderings, and $C(M) \ge b(M)$.

For strictness, every cell with $i \ge 2$ receives at least one unit beyond what was used to
dominate $Z_3$.
\begin{itemize}[nosep]
\item $j = 0$: $[x^i]U = 2^{i-1} \ge 2$.
\item $1 \le j < i$: the unused product $UV$ gives $[x^{i-j}]U \cdot [(xy)^j]V \ge 1$.
\item $j \ge i$: (ii) used only $W$ there. The term of $UW$ with $x$ taken from $U$, namely
  $[x]U \cdot [x^{i-1}y^j]W = \binom{j-1}{i-2} \ge 1$, is unused.
\end{itemize}

If $\max M \ge 2$, the decreasing ordering of $M$ is ballot with $\beta_1 = \max M \ge 2$. So
$C(M) \ge b(M) + 1$. Finally, the plane sector at $d = 3$ is $\{(0,0,0), (1,0,-1), (1,1,-2)\}$, where
$C = 1, 3, 1 = b$.
\end{proof}

\subsection{An explicit positive formula for $F_4$}

The same shape of argument needs a manifestly positive expression for the chamber series. At
$d = 4$ one exists. In chamber variables $x_1, x_2, x_3$ put $s = x_2x_3$ and $t = x_1x_2x_3$. The
Morin chamber series is
\[
  F_4 = \frac{N_4}{(1-2x_1)(1-2x_1x_2)(1-x_2-x_1x_2)(1-2t)(1-s-t)(1-x_3-t)(1-2s)} ,
\]
and its numerator factors completely:
\[
  N_4 = (1-x_1)(1-x_2)(1-x_1x_2)(1-x_3)(1-s)(1-t)(1-s-2t).
\]

\begin{lemma}\label{lem:f4}
\[
  F_4 = \Bigl(\tfrac12\,\tfrac{1}{1-2x_1} + \tfrac12\,\tfrac{1}{1-2s}
          + \frac{x_1^2 s}{(1-2x_1)(1-s-t)}\Bigr)
        \cdot \frac{1-x_1x_2}{1-2x_1x_2}\cdot\frac{1-x_2}{1-x_2-x_1x_2}
        \cdot \frac{1-t}{1-2t}\cdot\frac{1-x_3}{1-x_3-t} .
\]
Each of the last four factors is $1 + v/(1-u-v)$ with $u, v$ nonnegative, and the first factor is
a sum of nonnegative series. In particular $F_4$ has nonnegative coefficients (strong positivity
at $d = 4$).
\end{lemma}

\begin{proof}
Group the numerator factors $(1-x_1)(1-s)(1-s-2t)$ with the denominator factors
$(1-2x_1)(1-s-t)(1-2s)$, and write $a = x_1$, so that $t = as$. Then
\[
  (1-a)(1-s)(1-s-2as) = a^2 s (1-2s) + \tfrac12 (1-2a)(1-s-as) + \tfrac12 (1-s-as)(1-2s),
\]
which is checked by expanding both sides. Dividing by $(1-2a)(1-s-as)(1-2s)$ gives the first
factor, and the remaining four numerator factors pair with the remaining four denominator
factors as displayed.
\end{proof}

This expression was found by a small exact certificate search, and both identities are pinned in
tier~15.

\begin{theorem}\label{thm:d4}
The conclusion of Theorem~\ref{thm:small} holds for $d = 4$: $C(M) \ge b(M)$ for every zero-sum
$M$, with equality if and only if $\max M \le 1$.
\end{theorem}

\begin{proof}
\emph{Step 1: a lower bound for $F_4$ by supports.} Since $\tfrac12\,\frac{1}{1-2y} = \tfrac12 +
\frac{y}{1-2y}$, Lemma~\ref{lem:f4} reads $F_4 = (1 + X + S + Y)\prod_{i=1}^4 (1 + V_i)$ with
\[
  X = \tfrac{x_1}{1-2x_1},\ S = \tfrac{s}{1-2s},\ Y = \tfrac{x_1^2 s}{(1-2x_1)(1-s-t)},\
  V_1 = \tfrac{x_1x_2}{1-2x_1x_2},\ V_2 = \tfrac{x_1x_2}{1-x_2-x_1x_2},\
  V_3 = \tfrac{t}{1-2t},\ V_4 = \tfrac{t}{1-x_3-t}.
\]
Each factor has nonnegative integer coefficients, positive exactly on the following supports.
Write $(a,b,c)$ for an exponent vector.
- $X$: $b = c = 0 < a$.
- $S$: $a = 0 < b = c$.
- $Y$: $a \ge 2$, $b = c \ge 1$.
- $V_1$: $a = b \ge 1$, $c = 0$.
- $V_2$: $1 \le a \le b$, $c = 0$.
- $V_3$: $a = b = c \ge 1$.
- $V_4$: $1 \le a = b \le c$.

The coefficients are powers of $2$ or binomial coefficients $\binom mj$ with $0 \le j \le m$. For
example, $V_2 = \sum_m x_1x_2(x_2 + x_1x_2)^m$ has coefficient $\binom{m}{j}$ at $(1+j, 1+m, 0)$.
Expanding the product, $F_4$ is a sum of $64$ series with nonnegative integer coefficients, each
positive exactly on a Minkowski sum of these supports. Hence $F_4(\beta) \ge N(\beta)$, the number
of those $64$ supports that contain $\beta$.

\emph{Step 2: the comparison series.} The zero set of $F_4$ is
$\{a = 0,\ b \ne c\} \cup \{b = 0 < a, c\}$. The injective map in the text, extended by
$(0,0,c,-c) \mapsto (c,0,0,-c)$ and $(0,b,-b,0) \mapsto (b,0,-b,0)$, carries the ballot words landing
there bijectively onto the words landing on
\[
  T = \{b = c \ge 1 \le a\} \cup \{a = b \ge 1,\ c = 0\} \cup \{b = a + c,\ a, c \ge 1\},
\]
and every step permutes the entries of $\alpha$. So $Z_4 = P_4 - 1_{\text{zero set}} + 1_T$, where
$P_4 = \prod_j (1-x_j)^{-1}$, has packet sums exactly $b(M)$. Its coefficient is
$w = 1_{W_1} + 1_T$, where $W_1$ is the complement of the zero set.

\emph{Step 3: exact inclusions.} Let $D$ be the set of cells whose word is weakly decreasing with
$\alpha_1 \ge 2$. Every packet with $\max M \ge 2$ has its decreasing ordering in $D$. The
functions $N$ and $w + 1_D \le 3$ are defined by Presburger formulas. The inequality $N \ge w + 1_D$
is equivalent to three inclusions: for $k = 1, 2, 3$, $\{w + 1_D \ge k\}$ lies inside the set of
cells covered by at least $k$ of the $64$ supports. These inclusions, and $T, D \subseteq W_1$, are
decided exactly by the integer-set library isl (\texttt{chernpp.ballot.verify\_d4}, one second),
with no truncation.

Summing over a packet: $C(M) \ge \sum N \ge \sum (w + 1_D) = b(M) + [\max M \ge 2]$. The plane
sector at $d = 4$ consists of five packets, where $C = b$ is checked directly.
\end{proof}

The same three steps would apply at $d = 5$ given a positive expression for (a gauge of) $F_5$ and a
matching for its zero set. Strong positivity fails for the canonical $F_5$, so there the first step
must use a residue-null gauge.

\section{Zero insertion}

Appending a $0$ to $M$ adds a singleton block and multiplies the Chern monomial by $c_{\ell+1}$. For
a zero-free base $M_0$ write $C_z = C(M_0 \sqcup 0^z)$ and $b_z = b(M_0 \sqcup 0^z)$. On the plane
sector $b_z$ is a Kreweras number, a polynomial in $z$. For example $b(1,-1,0^z) = \binom{z+2}{2} = 1
+ 2z + \binom{z}{2}$.

\begin{conjecture}[zero insertion]\label{conj:zero}
For every zero-free $M_0$, the sequences $C_z$ and $C_z - b_z$ have nonnegative forward differences
of every order at $z = 0$:
\[
  C_z - b_z = \sum_{k \ge 0} N_k(M_0) \binom{z}{k}, \qquad N_k(M_0) \ge 0 .
\]
\end{conjecture}

This refines Conjecture~\ref{conj:ballot}, since every $M$ is some $M_0 \sqcup 0^z$. It holds on
all $4925$ zero-free bases available (\texttt{results/zero\_insertion.json}), for $C$ and for
$C - b$, at every order that fits in $d \le 7$. Where the differences terminate within range,
$C_z$ is a polynomial:
\[
  C(2,-2,0^z) = 2 + 5z + 4\binom z2 + \binom z3, \qquad
  C(3,-3,0^z) = 4 + 13 z + 16\binom z2 + 9\binom z3 + 2\binom z4 .
\]
The degree equals $\sum_{a > 0} a + \#\{a < 0\}$ in all four certified cases. That is too few to
conjecture.

\section{Where a proof has to go}

In the geometric model of \S\ref{sec:inputs}, a packet of charge $p$ reduces by (I3) only to
the curvilinear locus $C_{n,p+1}$ in $\mathbb{C}^{p+1}$. The plane case $p = 1$ is
Theorem~\ref{thm:plane}. Conjecture~\ref{conj:ballot} therefore says:
\begin{quote}
\emph{the integral of $m_\nu(E)$ over the curvilinear locus in $\mathbb{C}^{p+1}$ is at least the
count of ballot words, which is what the same integral gives in the plane, and exceeds it as soon
as $p \ge 2$.}
\end{quote}
Two routes suggest themselves.
\begin{enumerate}[nosep]
\item \emph{A degeneration.} Degenerate $C_{n,p+1}$ to a cycle containing a planar component with
  multiplicity one, plus effective components on which $m_\nu(\epsilon)$ integrates nonnegatively.
  Positivity does fail on some individual degeneration components~\cite{FS}, so the decomposition
  would have to be chosen with care.
\item \emph{A charge-2 Jucys--Murphy calculus.} Find the analogue, for the curvilinear component
  of the punctual Hilbert scheme of $\mathbb{C}^3$, of the Lehn--Sorger description. In such a
  calculus the excess $C(M) - b(M)$ should count a positive family of \emph{non-monotone} or
  \emph{branched} factorisations.
\end{enumerate}
The exact tables (\texttt{results/morin\_a\{6,7\}.json}, 20,725 coefficients) are available to test
either route.
